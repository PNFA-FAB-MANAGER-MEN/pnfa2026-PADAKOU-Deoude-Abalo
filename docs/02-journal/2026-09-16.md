from machine import Pin, PWM, Timer
import time

PWM_PIN = 5          # broche vers le filtre RC -> L IN du JQ-D503
SAMPLE_RATE = 8000    # doit correspondre à la conversion ffmpeg
PWM_FREQ = 62500      # fréquence porteuse, bien au-dessus du spectre audio

audio_pwm = PWM(Pin(PWM_PIN))
audio_pwm.freq(PWM_FREQ)

def play_bytes(data: bytes, sample_rate=SAMPLE_RATE):
    period_us = 1_000_000 // sample_rate
    for b in data:
        audio_pwm.duty_u16(b << 8)   # échantillon 8 bits -> duty 16 bits
        time.sleep_us(period_us)
    audio_pwm.duty_u16(0)

# Option A: audio embarqué directement dans le programme
son1 = b'\x80\x82\x85\x7f...'   # vos octets bruts (générés depuis le .raw)
play_bytes(son1)
